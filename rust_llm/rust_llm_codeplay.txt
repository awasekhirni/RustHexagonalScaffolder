```rust
//! Configuration for the LLM. Mirrors `LLMConfig` from the Python version.
//!
//! Centralises every architectural knob (Ch. 3) and validation rule so the
//! rest of the codebase can treat the model shape as data, not code.

use serde::{Deserialize, Serialize};

/// Complete hyperparameter set for a decoder-only Transformer LLM.
///
/// See the Python `LLMConfig` docstring for the field-by-field rationale.
/// In Rust we split construction into `LLMConfig::tiny()` and a builder-style
/// `with_*` API to keep struct literal sites readable.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct LLMConfig {
    pub vocab_size: usize,
    pub max_seq_len: usize,
    pub hidden_size: usize,
    pub num_layers: usize,
    pub num_heads: usize,
    pub num_kv_heads: usize,
    pub intermediate_size: usize,
    pub use_rope: bool,
    pub rope_theta: f64,
    pub use_swiglu: bool,
    pub use_rms_norm: bool,
    pub tie_embeddings: bool,
    pub initializer_range: f64,
    pub dropout: f64,
    pub attention_dropout: f64,
}

impl LLMConfig {
    /// A small default config suitable for CPU training in minutes.
    pub fn tiny(vocab_size: usize) -> Self {
        let hidden_size = 128;
        let num_heads = 4;
        let num_kv_heads = 2;

        // Llama-style FFN ratio for SwiGLU: 8/3 * d_model, rounded up to 64.
        let raw = (8 * hidden_size + 2) / 3;
        let intermediate_size = ((raw + 63) / 64) * 64;

        Self {
            vocab_size,
            max_seq_len: 64,
            hidden_size,
            num_layers: 4,
            num_heads,
            num_kv_heads,
            intermediate_size,
            use_rope: true,
            rope_theta: 10_000.0,
            use_swiglu: true,
            use_rms_norm: true,
            tie_embeddings: true,
            initializer_range: 0.02,
            dropout: 0.0,
            attention_dropout: 0.0,
        }
    }

    /// Per-head dimension `d_k = d_model / num_heads`.
    pub fn head_dim(&self) -> usize {
        self.hidden_size / self.num_heads
    }

    /// Number of query heads that share each KV head.
    pub fn num_kv_groups(&self) -> usize {
        self.num_heads / self.num_kv_heads
    }

    /// Validate invariants.
    pub fn validate(&self) -> anyhow::Result<()> {
        anyhow::ensure!(
            self.hidden_size % self.num_heads == 0,
            "hidden_size must be divisible by num_heads"
        );
        anyhow::ensure!(
            self.num_heads % self.num_kv_heads == 0,
            "num_heads must be divisible by num_kv_heads for GQA"
        );
        anyhow::ensure!(
            self.head_dim() % 2 == 0,
            "head_dim must be even for RoPE"
        );
        Ok(())
    }
}
```

## `src/tokenizer.rs`

```rust
//! Byte-level BPE tokenizer (Chapter 2).
//!
//! Implements the canonical BPE algorithm: start with the 256 raw byte
//! symbols, repeatedly merge the most frequent adjacent pair, assign a new
//! ID. Encoding applies merges in the order they were learned.

use std::collections::HashMap;
use std::fs;
use std::path::Path;

use anyhow::{anyhow, Result};
use serde::{Deserialize, Serialize};

/// A pair of token IDs.
type Pair = (u32, u32);

/// Byte-level BPE tokenizer with a small set of reserved chat tokens.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct BpeTokenizer {
    /// token_id -> raw bytes
    vocab: HashMap<u32, Vec<u8>>,
    /// (left, right) -> merged id
    merges: HashMap<Pair, u32>,
    /// string -> id
    specials: HashMap<String, u32>,
    target_vocab_size: usize,
    trained: bool,
}

impl BpeTokenizer {
    /// Default special tokens, starting at ID 256.
    pub fn default_specials() -> Vec<(&'static str, u32)> {
        vec![
            ("<|endoftext|>", 256),
            ("<|pad|>", 257),
            ("<|user|>", 258),
            ("<|assistant|>", 259),
        ]
    }

    /// Create a fresh, untrained tokenizer.
    pub fn new(target_vocab_size: usize) -> Self {
        let mut vocab = HashMap::new();
        for i in 0u32..256 {
            vocab.insert(i, vec![i as u8]);
        }
        let mut specials = HashMap::new();
        let mut next = 256u32;
        for (tok, _) in Self::default_specials() {
            specials.insert(tok.to_string(), next);
            vocab.insert(next, tok.as_bytes().to_vec());
            next += 1;
        }
        Self {
            vocab,
            merges: HashMap::new(),
            specials,
            target_vocab_size,
            trained: false,
        }
    }

    /// Learn merges from `text` until `target_vocab_size` is reached.
    pub fn train(&mut self, text: &str, verbose: bool) -> Result<()> {
        let mut tokens: Vec<u32> = text.bytes().map(|b| b as u32).collect();

        while (self.vocab.len() as usize) < self.target_vocab_size {
            // Count adjacent pairs.
            let mut counts: HashMap<Pair, u32> = HashMap::new();
            for w in tokens.windows(2) {
                *counts.entry((w[0], w[1])).or_insert(0) += 1;
            }
            let Some((&best_pair, &freq)) = counts.iter().max_by_key(|(_, &v)| v) else {
                break;
            };
            if freq < 2 {
                break; // no repeated pair; further merging is useless.
            }

            let new_id = self.vocab.len() as u32;
            self.vocab.insert(new_id, {
                let mut v = self.vocab[&best_pair.0].clone();
                v.extend_from_slice(&self.vocab[&best_pair.1]);
                v
            });
            self.merges.insert(best_pair, new_id);

            // Rewrite tokens, merging every occurrence.
            let mut new_tokens = Vec::with_capacity(tokens.len());
            let mut i = 0;
            while i < tokens.len() {
                if i + 1 < tokens.len()
                    && tokens[i] == best_pair.0
                    && tokens[i + 1] == best_pair.1
                {
                    new_tokens.push(new_id);
                    i += 2;
                } else {
                    new_tokens.push(tokens[i]);
                    i += 1;
                }
            }
            tokens = new_tokens;

            if verbose && self.vocab.len() % 100 == 0 {
                println!(
                    "[BPE] vocab={} merged ({}, {}) -> {} (freq={})",
                    self.vocab.len(),
                    best_pair.0,
                    best_pair.1,
                    new_id,
                    freq
                );
            }
        }
        self.trained = true;
        Ok(())
    }

    /// Encode a string to a sequence of token IDs.
    pub fn encode(&self, text: &str, allow_special: bool) -> Vec<u32> {
        if !self.trained {
            return text.bytes().map(|b| b as u32).collect();
        }

        // Handle special tokens: replace each with a sentinel byte.
        if allow_special {
            let mut current = text.to_string();
            let mut sentinels: Vec<(String, u32)> = Vec::new();
            for (tok, id) in &self.specials {
                let sentinel = format!("\u{1}{}\u{1}", id);
                if current.contains(tok) {
                    current = current.replace(tok, &sentinel);
                    sentinels.push((sentinel, *id));
                }
            }
            // Encode segments split by sentinel.
            let mut out = Vec::new();
            let mut remaining = current.as_str();
            while !remaining.is_empty() {
                // Find earliest sentinel.
                let mut earliest: Option<(usize, &(String, u32))> = None;
                for s in &sentinels {
                    if let Some(pos) = remaining.find(&s.0) {
                        if earliest.map_or(true, |(p, _)| pos < p) {
                            earliest = Some((pos, s));
                        }
                    }
                }
                match earliest {
                    Some((pos, (sentinel, id))) => {
                        out.extend(self.bpe_encode(&remaining[..pos]));
                        out.push(*id);
                        remaining = &remaining[pos + sentinel.len()..];
                    }
                    None => {
                        out.extend(self.bpe_encode(remaining));
                        break;
                    }
                }
            }
            return out;
        }

        self.bpe_encode(text)
    }

    fn bpe_encode(&self, text: &str) -> Vec<u32> {
        let mut tokens: Vec<u32> = text.bytes().map(|b| b as u32).collect();

        loop {
            if tokens.len() < 2 {
                break;
            }
            // Find the pair with the smallest merge-rank (earliest learned).
            let mut best: Option<(u32, Pair)> = None;
            for w in tokens.windows(2) {
                if let Some(&new_id) = self.merges.get(&(w[0], w[1])) {
                    if best.map_or(true, |(b, _)| new_id < b) {
                        best = Some((new_id, (w[0], w[1])));
                    }
                }
            }
            let Some((new_id, pair)) = best else { break };

            let mut merged = Vec::with_capacity(tokens.len());
            let mut i = 0;
            while i < tokens.len() {
                if i + 1 < tokens.len() && tokens[i] == pair.0 && tokens[i + 1] == pair.1 {
                    merged.push(new_id);
                    i += 2;
                } else {
                    merged.push(tokens[i]);
                    i += 1;
                }
            }
            tokens = merged;
        }

        tokens
    }

    /// Decode token IDs back to a UTF-8 string.
    pub fn decode(&self, ids: &[u32], skip_special: bool) -> String {
        let mut buf = Vec::new();
        for &id in ids {
            if let Some(tok) = self.specials.iter().find(|(_, &v)| v == id).map(|(k, _)| k) {
                if skip_special {
                    continue;
                }
                buf.extend_from_slice(tok.as_bytes());
                continue;
            }
            if let Some(b) = self.vocab.get(&id) {
                buf.extend_from_slice(b);
            }
        }
        String::from_utf8_lossy(&buf).into_owned()
    }

    pub fn vocab_size(&self) -> usize {
        self.vocab.len()
    }

    pub fn pad_id(&self) -> u32 {
        *self.specials.get("<|pad|>").unwrap()
    }

    pub fn eos_id(&self) -> u32 {
        *self.specials.get("<|endoftext|>").unwrap()
    }

    /// Save as JSON.
    pub fn save<P: AsRef<Path>>(&self, path: P) -> Result<()> {
        let json = serde_json::to_string(self)?;
        fs::write(path, json)?;
        Ok(())
    }

    /// Load from JSON.
    pub fn load<P: AsRef<Path>>(path: P) -> Result<Self> {
        let json = fs::read_to_string(path)?;
        let tok: BpeTokenizer = serde_json::from_str(&json)?;
        Ok(tok)
    }
}

// Serialisation helper for HashMap<Pair, u32> — serde cannot natively
// handle tuple keys in JSON, so we stringify them.
impl BpeTokenizer {
    pub fn to_json_safe(&self) -> Result<String> {
        // Use a "merges_list" representation.
        let merges: Vec<(u32, u32, u32)> = self
            .merges
            .iter()
            .map(|(&(a, b), &c)| (a, b, c))
            .collect();
        let vocab: Vec<(u32, Vec<u8>)> = self.vocab.iter().map(|(&k, v)| (k, v.clone())).collect();
        let specials: Vec<(String, u32)> =
            self.specials.iter().map(|(k, v)| (k.clone(), *v)).collect();
        let obj = serde_json::json!({
            "vocab": vocab,
            "merges": merges,
            "specials": specials,
            "target_vocab_size": self.target_vocab_size,
            "trained": self.trained,
        });
        Ok(serde_json::to_string(&obj)?)
    }
}

/// Parse the JSON-safe representation created by `to_json_safe`.
pub fn bpe_from_json(json: &str) -> Result<BpeTokenizer> {
    let v: serde_json::Value = serde_json::from_str(json)?;
    let mut vocab = HashMap::new();
    for pair in v["vocab"].as_array().ok_or_else(|| anyhow!("vocab missing"))? {
        let id = pair[0].as_u64().unwrap() as u32;
        let bytes: Vec<u8> = pair[1]
            .as_array()
            .unwrap()
            .iter()
            .map(|b| b.as_u64().unwrap() as u8)
            .collect();
        vocab.insert(id, bytes);
    }
    let mut merges = HashMap::new();
    for triple in v["merges"].as_array().unwrap() {
        let a = triple[0].as_u64().unwrap() as u32;
        let b = triple[1].as_u64().unwrap() as u32;
        let c = triple[2].as_u64().unwrap() as u32;
        merges.insert((a, b), c);
    }
    let mut specials = HashMap::new();
    for pair in v["specials"].as_array().unwrap() {
        specials.insert(pair[0].as_str().unwrap().to_string(), pair[1].as_u64().unwrap() as u32);
    }
    Ok(BpeTokenizer {
        vocab,
        merges,
        specials,
        target_vocab_size: v["target_vocab_size"].as_u64().unwrap() as usize,
        trained: v["trained"].as_bool().unwrap(),
    })
}
```

```rust
//! The LLM: RoPE, RMSNorm, GQA attention with KV cache, SwiGLU, Transformer
//! block, and full decoder stack (Chapters 1 & 3).

use anyhow::Result;
use candle_core::{DType, Device, IndexOp, Module, Tensor, D};
use candle_nn::{embedding, linear, linear_no_bias, Embedding, Linear, VarBuilder};

use crate::config::LLMConfig;

// =========================================================================
// RMSNorm / LayerNorm
// =========================================================================

/// Root-mean-square layer norm (Ch. 3, "The Transformer Block").
#[derive(Debug, Clone)]
pub struct RmsNorm {
    weight: Tensor,
    eps: f64,
}

impl RmsNorm {
    pub fn new(dim: usize, eps: f64, vb: VarBuilder) -> Result<Self> {
        let weight = vb.get(dim, "weight")?;
        Ok(Self { weight, eps })
    }
}

impl Module for RmsNorm {
    fn forward(&self, x: &Tensor) -> Result<Tensor> {
        let dtype = x.dtype();
        let x32 = x.to_dtype(DType::F32)?;
        let ms = x32.sqr()?.mean_keepdim(D::Minus1)?;
        let rms = (ms + self.eps)?.sqrt()?.recip()?;
        let normed = x32.broadcast_mul(&rms)?;
        normed.to_dtype(dtype)?.broadcast_mul(&self.weight)
    }
}

// =========================================================================
// RoPE (Chapter 3, "Positional Embeddings (RoPE)")
// =========================================================================

/// Precomputed rotary positional embedding tables.
#[derive(Debug, Clone)]
pub struct Rope {
    cos: Tensor, // (max_seq, head_dim)
    sin: Tensor,
}

impl Rope {
    pub fn new(head_dim: usize, max_seq_len: usize, theta: f64, dev: &Device) -> Result<Self> {
        assert!(head_dim % 2 == 0);
        let half = head_dim / 2;
        let inv_freq: Vec<f32> = (0..half)
            .map(|i| 1.0 / (theta as f32).powf((2.0 * i as f32) / head_dim as f32))
            .collect();
        let inv_freq = Tensor::from_vec(inv_freq, (half,), dev)?;
        let positions: Vec<f32> = (0..max_seq_len).map(|i| i as f32).collect();
        let positions = Tensor::from_vec(positions, (max_seq_len,), dev)?;
        // (S, 1) * (1, half) -> (S, half)
        let freqs = positions
            .unsqueeze(1)?
            .broadcast_mul(&inv_freq.unsqueeze(0)?)?;
        // Duplicate to (S, head_dim)
        let emb = Tensor::cat(&[&freqs, &freqs], D::Minus1)?;
        let cos = emb.cos()?;
        let sin = emb.sin()?;
        Ok(Self { cos, sin })
    }

    /// Apply RoPE to q,k of shape `(B, H, S, D)`, given the absolute position
    /// offset of the first row.
    pub fn apply(&self, q: &Tensor, k: &Tensor, offset: usize) -> Result<(Tensor, Tensor)> {
        let s = q.dim(2)?;
        let cos = self.cos.narrow(0, offset, s)?;
        let sin = self.sin.narrow(0, offset, s)?;
        // (S, D) -> (1, 1, S, D)
        let cos = cos.unsqueeze(0)?.unsqueeze(0)?;
        let sin = sin.unsqueeze(0)?.unsqueeze(0)?;

        let q_rot = rotate_half(q)?;
        let k_rot = rotate_half(k)?;

        let q_out = q
            .broadcast_mul(&cos)?
            .add(&q_rot.broadcast_mul(&sin)?)?;
        let k_out = k
            .broadcast_mul(&cos)?
            .add(&k_rot.broadcast_mul(&sin)?)?;
        Ok((q_out, k_out))
    }
}

fn rotate_half(x: &Tensor) -> Result<Tensor> {
    let d = x.dim(D::Minus1)?;
    let half = d / 2;
    let x1 = x.narrow(D::Minus1, 0, half)?;
    let x2 = x.narrow(D::Minus1, half, half)?;
    Tensor::cat(&[&x2.neg()?, &x1], D::Minus1)
}

// =========================================================================
// KV Cache
// =========================================================================

/// Per-layer key/value cache for autoregressive decoding.
#[derive(Debug, Clone)]
pub struct KvCache {
    pub k: Tensor, // (B, num_kv_heads, S_past, head_dim)
    pub v: Tensor,
}

// =========================================================================
// Multi-Head / Grouped-Query Attention (Chapter 3)
// =========================================================================

/// Causal self-attention with GQA + RoPE + KV cache.
pub struct MultiHeadAttention {
    q_proj: Linear,
    k_proj: Linear,
    v_proj: Linear,
    o_proj: Linear,
    num_heads: usize,
    num_kv_heads: usize,
    head_dim: usize,
    num_kv_groups: usize,
    rope: Option<Rope>,
    rope_theta: f64,
}

impl MultiHeadAttention {
    pub fn new(cfg: &LLMConfig, vb: VarBuilder) -> Result<Self> {
        let h = cfg.hidden_size;
        let q_proj = linear_no_bias(h, cfg.num_heads * cfg.head_dim(), vb.pp("q_proj"))?;
        let k_proj = linear_no_bias(h, cfg.num_kv_heads * cfg.head_dim(), vb.pp("k_proj"))?;
        let v_proj = linear_no_bias(h, cfg.num_kv_heads * cfg.head_dim(), vb.pp("v_proj"))?;
        let o_proj = linear_no_bias(cfg.num_heads * cfg.head_dim(), h, vb.pp("o_proj"))?;

        let rope = if cfg.use_rope {
            Some(Rope::new(
                cfg.head_dim(),
                cfg.max_seq_len * 4, // headroom for generation beyond training
                cfg.rope_theta,
                vb.device(),
            )?)
        } else {
            None
        };

        Ok(Self {
            q_proj,
            k_proj,
            v_proj,
            o_proj,
            num_heads: cfg.num_heads,
            num_kv_heads: cfg.num_kv_heads,
            head_dim: cfg.head_dim(),
            num_kv_groups: cfg.num_kv_groups(),
            rope,
            rope_theta: cfg.rope_theta,
        })
    }

    /// Forward.
    ///
    /// * `x`     — (B, S, d_model)
    /// * `cache` — optional past (k, v)
    pub fn forward(&self, x: &Tensor, cache: Option<&KvCache>) -> Result<(Tensor, KvCache)> {
        let (b, s, _) = x.dims3()?;
        let past = cache.map(|c| c.k.dim(2)).unwrap_or(0);

        // --- 1. Projections ---
        let q = self.q_proj.forward(x)?; // (B, S, H*D)
        let k = self.k_proj.forward(x)?; // (B, S, Hkv*D)
        let v = self.v_proj.forward(x)?; // (B, S, Hkv*D)

        let q = q
            .reshape((b, s, self.num_heads, self.head_dim))?
            .transpose(1, 2)?; // (B, H, S, D)
        let k = k
            .reshape((b, s, self.num_kv_heads, self.head_dim))?
            .transpose(1, 2)?;
        let v = v
            .reshape((b, s, self.num_kv_heads, self.head_dim))?
            .transpose(1, 2)?;

        // --- 2. RoPE ---
        let (q, k) = if let Some(rope) = &self.rope {
            rope.apply(&q, &k, past)?
        } else {
            (q, k)
        };

        // --- 3. KV cache ---
        let (k, v) = match cache {
            Some(c) => {
                let k = Tensor::cat(&[&c.k, &k], 2)?;
                let v = Tensor::cat(&[&c.v, &v], 2)?;
                (k, v)
            }
            None => (k, v),
        };
        let new_cache = KvCache {
            k: k.clone(),
            v: v.clone(),
        };

        let s_kv = k.dim(2)?;

        // --- 4. GQA: repeat KV heads to match Q heads ---
        let (k, v) = if self.num_kv_groups > 1 {
            let k = k
                .unsqueeze(2)? // (B, Hkv, 1, S, D)
                .broadcast_as((b, self.num_kv_heads, self.num_kv_groups, s_kv, self.head_dim))?
                .reshape((b, self.num_kv_heads * self.num_kv_groups, s_kv, self.head_dim))?;
            let v = v
                .unsqueeze(2)?
                .broadcast_as((b, self.num_kv_heads, self.num_kv_groups, s_kv, self.head_dim))?
                .reshape((b, self.num_kv_heads * self.num_kv_groups, s_kv, self.head_dim))?;
            (k, v)
        } else {
            (k, v)
        };

        // --- 5. Attention scores ---
        // (B, H, S, D) @ (B, H, D, S_kv) -> (B, H, S, S_kv)
        let scale = 1.0 / (self.head_dim as f64).sqrt();
        let att = (q.matmul(&k.transpose(2, 3)?)? * scale)?;

        // Causal mask (only when no cache — a single cached step already
        // sees only past positions by construction).
        let att = if s > 1 && past == 0 {
            let mask = make_causal_mask(s, x.device(), att.dtype())?;
            att.broadcast_add(&mask)?
        } else if past > 0 && s > 1 {
            // Prefill + multiple queries with cache: need causal + offset.
            let mask = make_causal_mask_offset(s, s_kv, past, x.device(), att.dtype())?;
            att.broadcast_add(&mask)?
        } else {
            att
        };

        let att = candle_nn::ops::softmax_last_dim(&att)?;
        let out = att.matmul(&v)?; // (B, H, S, D)

        // --- 6. Merge heads and output projection ---
        let out = out
            .transpose(1, 2)? // (B, S, H, D)
            .reshape((b, s, self.num_heads * self.head_dim))?;
        let out = self.o_proj.forward(&out)?;
        Ok((out, new_cache))
    }

    #[allow(dead_code)]
    pub fn rope_theta(&self) -> f64 {
        self.rope_theta
    }
}

fn make_causal_mask(s: usize, dev: &Device, dtype: DType) -> Result<Tensor> {
    // Upper triangular -inf above the diagonal.
    let mut data = vec![0f32; s * s];
    for i in 0..s {
        for j in (i + 1)..s {
            data[i * s + j] = f32::NEG_INFINITY;
        }
    }
    Ok(Tensor::from_vec(data, (1, 1, s, s), dev)?.to_dtype(dtype)?)
}

fn make_causal_mask_offset(
    s_q: usize,
    s_kv: usize,
    offset: usize,
    dev: &Device,
    dtype: DType,
) -> Result<Tensor> {
    // Query row `i` corresponds to absolute position `offset + i`.
    // Allow attention to keys at absolute positions 0..=offset+i.
    let mut data = vec![0f32; s_q * s_kv];
    for i in 0..s_q {
        for j in 0..s_kv {
            if j > offset + i {
                data[i * s_kv + j] = f32::NEG_INFINITY;
            }
        }
    }
    Ok(Tensor::from_vec(data, (1, 1, s_q, s_kv), dev)?.to_dtype(dtype)?)
}

/// Position-wise feedforward (Ch. 3), SwiGLU variant.
pub struct FeedForward {
    w1: Linear, // gate
    w3: Linear, // value
    w2: Linear, // down
}

impl FeedForward {
    pub fn new(cfg: &LLMConfig, vb: VarBuilder) -> Result<Self> {
        assert!(cfg.use_swiglu, "Rust port only implements SwiGLU for brevity");
        let w1 = linear_no_bias(cfg.hidden_size, cfg.intermediate_size, vb.pp("w1"))?;
        let w3 = linear_no_bias(cfg.hidden_size, cfg.intermediate_size, vb.pp("w3"))?;
        let w2 = linear_no_bias(cfg.intermediate_size, cfg.hidden_size, vb.pp("w2"))?;
        Ok(Self { w1, w3, w2 })
    }
}

impl Module for FeedForward {
    fn forward(&self, x: &Tensor) -> Result<Tensor> {
        let gate = self.w1.forward(x)?.silu()?;
        let value = self.w3.forward(x)?;
        self.w2.forward(&(gate * value)?)
    }
}

// Transformer Block

/// Pre-norm Transformer block: `x = x + Attn(Norm(x)); x = x + FFN(Norm(x))`.
pub struct TransformerBlock {
    attn_norm: RmsNorm,
    attn: MultiHeadAttention,
    ffn_norm: RmsNorm,
    ffn: FeedForward,
}

impl TransformerBlock {
    pub fn new(cfg: &LLMConfig, vb: VarBuilder) -> Result<Self> {
        let attn_norm = RmsNorm::new(cfg.hidden_size, 1e-6, vb.pp("attn_norm"))?;
        let attn = MultiHeadAttention::new(cfg, vb.pp("attn"))?;
        let ffn_norm = RmsNorm::new(cfg.hidden_size, 1e-6, vb.pp("ffn_norm"))?;
        let ffn = FeedForward::new(cfg, vb.pp("ffn"))?;
        Ok(Self {
            attn_norm,
            attn,
            ffn_norm,
            ffn,
        })
    }

    pub fn forward(&self, x: &Tensor, cache: Option<&KvCache>) -> Result<(Tensor, KvCache)> {
        let h = self.attn_norm.forward(x)?;
        let (h, new_cache) = self.attn.forward(&h, cache)?;
        let x = (x + &h)?;

        let h = self.ffn_norm.forward(&x)?;
        let h = self.ffn.forward(&h)?;
        let x = (x + &h)?;
        Ok((x, new_cache))
    }
}

/// Decoder-only Transformer LM with weight-tied LM head.
pub struct Llm {
    pub cfg: LLMConfig,
    embed: Embedding,
    pos_embed: Option<Embedding>, // only when use_rope == false
    blocks: Vec<TransformerBlock>,
    final_norm: RmsNorm,
    lm_head: Option<Linear>, // None when tying
    pub embed_weight: Tensor, // shared tensor for weight tying
}

impl Llm {
    pub fn new(cfg: LLMConfig, vb: VarBuilder) -> Result<Self> {
        cfg.validate()?;
        let embed = embedding(cfg.vocab_size, cfg.hidden_size, vb.pp("embed_tokens"))?;
        let embed_weight = embed.embeddings().clone();

        let pos_embed = if !cfg.use_rope {
            Some(embedding(cfg.max_seq_len, cfg.hidden_size, vb.pp("pos_embed"))?)
        } else {
            None
        };

        let mut blocks = Vec::with_capacity(cfg.num_layers);
        for i in 0..cfg.num_layers {
            blocks.push(TransformerBlock::new(&cfg, vb.pp(format!("block_{i}")))?);
        }

        let final_norm = RmsNorm::new(cfg.hidden_size, 1e-6, vb.pp("final_norm"))?;

        let lm_head = if cfg.tie_embeddings {
            None
        } else {
            Some(linear_no_bias(
                cfg.hidden_size,
                cfg.vocab_size,
                vb.pp("lm_head"),
            )?)
        };

        Ok(Self {
            cfg,
            embed,
            pos_embed,
            blocks,
            final_norm,
            lm_head,
            embed_weight,
        })
    }

    /// Forward pass. Returns logits of shape (B, S, vocab).
    ///
    /// * `input_ids` — (B, S) u32
    /// * `caches`    — optional per-layer past KVs
    /// * `return_caches` — if true, return the updated caches
    pub fn forward(
        &self,
        input_ids: &Tensor,
        caches: Option<&[KvCache]>,
        return_caches: bool,
    ) -> Result<(Tensor, Option<Vec<KvCache>>)> {
        let (b, s) = input_ids.dims2()?;
        let dev = input_ids.device();

        // 1. Token embeddings.
        let mut x = self.embed.forward(input_ids)?;

        // 2. Positional embeddings if not RoPE.
        let past = caches.map(|c| c[0].k.dim(2)).unwrap_or(0);
        if let Some(pos) = &self.pos_embed {
            let idx: Vec<u32> = (past..past + s).map(|i| i as u32).collect();
            let idx = Tensor::from_vec(idx, (s,), dev)?;
            let p = pos.forward(&idx)?;
            x = x.broadcast_add(&p.unsqueeze(0)?)?;
        }

        // 3. Transformer blocks.
        let mut new_caches = Vec::with_capacity(self.blocks.len());
        for (i, block) in self.blocks.iter().enumerate() {
            let c = caches.map(|cs| &cs[i]);
            let (nx, nc) = block.forward(&x, c)?;
            x = nx;
            new_caches.push(nc);
        }

        // 4. Final norm + LM head.
        let x = self.final_norm.forward(&x)?;
        let logits = match &self.lm_head {
            Some(head) => head.forward(&x)?,
            None => {
                // Weight tying: x @ embed_weight^T
                let w = self.embed_weight.transpose(0, 1)?; // (d, vocab)
                x.matmul(&w)?
            }
        };

        let _ = b;
        if return_caches {
            Ok((logits, Some(new_caches)))
        } else {
            Ok((logits, None))
        }
    }

    /// Trainable parameter count.
    pub fn num_parameters(&self) -> usize {
        let mut n = 0;
        for p in self.blocks.iter() {
            // Approximation; count linear weights only.
            n += params_of_block(p);
        }
        n += self.embed_weight.elem_count();
        n
    }
}

fn params_of_block(_b: &TransformerBlock) -> usize {
    // Conservative: skip for brevity in the demo.
    0
}

// Helper trait to expose embedding tensor (candle's Embedding is opaque).
trait EmbeddingAccess {
    fn embeddings(&self) -> &Tensor;
}
impl EmbeddingAccess for Embedding {
    fn embeddings(&self) -> &Tensor {
        // In candle-nn, Embedding holds `embeddings: Tensor`.
        // We cannot access it directly; use Module to get output for a
        // known input and infer the weight instead.
        // For the Rust port we rely on `linear` heads sharing memory only
        // via the VarMap path names. See the note in the setup guide.
        unreachable!("see note in setup guide about weight tying in candle-nn")
    }
}
```

> **Note on weight tying in Candle:** Candle's `Embedding` and `Linear` are opaque wrappers, so accessing their inner `Tensor` requires either `VarMap::get` or a small wrapper type. In a production port you would implement a custom `TiedEmbedding` module holding a `Tensor` directly. The above scaffold illustrates the intent; for a runnable demo you can disable tying (`cfg.tie_embeddings = false`) and use the standard `Linear` head — behaviour is otherwise identical.

```rust
//! Autoregressive text generation with sampling (Chapters 3 & 6).

use anyhow::Result;
use candle_core::{DType, Device, Tensor, D};
use candle_nn::ops::softmax_last_dim;

use crate::model::{KvCache, Llm};
use crate::tokenizer::BpeTokenizer;

/// Sampling parameters.
#[derive(Debug, Clone)]
pub struct SamplingParams {
    pub max_new_tokens: usize,
    pub temperature: f64,
    pub top_k: usize,
    pub top_p: f64,
    pub repetition_penalty: f64,
    pub do_sample: bool,
    pub use_cache: bool,
    pub stop_on_eos: bool,
}

impl Default for SamplingParams {
    fn default() -> Self {
        Self {
            max_new_tokens: 32,
            temperature: 1.0,
            top_k: 0,
            top_p: 1.0,
            repetition_penalty: 1.0,
            do_sample: true,
            use_cache: true,
            stop_on_eos: false,
        }
    }
}

/// Handles decoding for a trained `Llm`.
pub struct TextGenerator<'a> {
    pub model: &'a Llm,
    pub tokenizer: &'a BpeTokenizer,
}

impl<'a> TextGenerator<'a> {
    pub fn new(model: &'a Llm, tokenizer: &'a BpeTokenizer) -> Self {
        Self { model, tokenizer }
    }

    /// Generate a continuation for `prompt`.
    pub fn generate(&self, prompt: &str, p: &SamplingParams) -> Result<String> {
        let dev = Device::Cpu; // or Cuda(0)
        let mut ids: Vec<u32> = self.tokenizer.encode(prompt, true);

        if ids.is_empty() {
            ids.push(self.tokenizer.eos_id());
        }

        // Tensor shape (1, S)
        let mut input = Tensor::from_vec(
            ids.iter().map(|&x| x as u32).collect::<Vec<_>>(),
            (1, ids.len()),
            &dev,
        )?;

        let mut caches: Option<Vec<KvCache>> = None;

        for _ in 0..p.max_new_tokens {
            let input_step = if p.use_cache && caches.is_some() {
                // Feed only the newest token.
                let last = input.narrow(1, input.dim(1)? - 1, 1)?;
                last
            } else {
                input.clone()
            };

            let (logits, new_caches) = self.model.forward(&input_step, caches.as_deref(), true)?;
            if p.use_cache {
                caches = new_caches;
            }

            let s = logits.dim(1)?;
            let next_logits = logits.narrow(1, s - 1, 1)?.squeeze(1)?; // (1, V)

            let next_id = self.sample(&next_logits, &input, p)?;
            let next_t = Tensor::from_vec(vec![next_id], (1, 1), &dev)?;
            input = Tensor::cat(&[&input, &next_t], 1)?;

            if p.stop_on_eos && next_id == self.tokenizer.eos_id() {
                break;
            }
            if input.dim(1)? >= self.model.cfg.max_seq_len {
                break;
            }
        }

        let out_ids: Vec<u32> = input
            .to_dtype(DType::U32)?
            .flatten_all()?
            .to_vec1()?;
        Ok(self.tokenizer.decode(&out_ids, true))
    }

    fn sample(
        &self,
        logits: &Tensor,   // (1, V)
        ctx: &Tensor,      // (1, S) — for repetition penalty
        p: &SamplingParams,
    ) -> Result<u32> {
        let mut l = logits.to_dtype(DType::F32)?.squeeze(0)?; // (V,)

        // 1. Repetition penalty
        if p.repetition_penalty != 1.0 {
            let seen: Vec<u32> = ctx
                .to_dtype(DType::U32)?
                .flatten_all()?
                .to_vec1()?;
            let mut unique: Vec<u32> = seen.clone();
            unique.sort_unstable();
            unique.dedup();
            let l_vec: Vec<f32> = l.to_vec1()?;
            let mut l_new = l_vec.clone();
            for &t in &unique {
                let idx = t as usize;
                if idx >= l_new.len() {
                    continue;
                }
                if l_new[idx] > 0.0 {
                    l_new[idx] /= p.repetition_penalty as f32;
                } else {
                    l_new[idx] *= p.repetition_penalty as f32;
                }
            }
            l = Tensor::from_vec(l_new, (l.dim(0)?,), logits.device())?;
        }

        // 2. Greedy?
        if !p.do_sample || p.temperature == 0.0 {
            return Ok(l.argmax(0)?.to_scalar::<u32>()?);
        }

        // 3. Temperature
        let l = (&l / p.temperature)?;

        // 4. Top-k
        let v = l.dim(0)?;
        let l = if p.top_k > 0 && p.top_k < v {
            let (vals, _) = topk(&l, p.top_k)?;
            let cutoff = vals[p.top_k - 1];
            let l_vec: Vec<f32> = l.to_vec1()?;
            let filtered: Vec<f32> = l_vec
                .iter()
                .map(|&x| if x < cutoff { f32::NEG_INFINITY } else { x })
                .collect();
            Tensor::from_vec(filtered, (v,), logits.device())?
        } else {
            l
        };

        // 5. Top-p (nucleus)
        let l = if p.top_p < 1.0 {
            let l_vec: Vec<f32> = l.to_vec1()?;
            let mut indexed: Vec<(usize, f32)> = l_vec.iter().copied().enumerate().collect();
            indexed.sort_by(|a, b| b.1.partial_cmp(&a.1).unwrap_or(std::cmp::Ordering::Equal));
            let mut cum = 0.0f32;
            let mut keep = vec![false; v];
            for (i, (orig_i, val)) in indexed.iter().enumerate() {
                let _ = i;
                let p_i = val.exp();
                cum += p_i;
                keep[*orig_i] = true;
                if cum > 1.0 {
                    break;
                }
            }
            let filtered: Vec<f32> = l_vec
                .iter()
                .enumerate()
                .map(|(i, &x)| if keep[i] { x } else { f32::NEG_INFINITY })
                .collect();
            Tensor::from_vec(filtered, (v,), logits.device())?
        } else {
            l
        };

        // 6. Sample via multinomial
        let probs = softmax_last_dim(&l.unsqueeze(0)?)?.squeeze(0)?;
        let mut rng = rand::thread_rng();
        let probs_vec: Vec<f32> = probs.to_vec1()?;
        Ok(sample_categorical(&probs_vec, &mut rng))
    }
}

fn topk(x: &Tensor, k: usize) -> Result<(Vec<f32>, Vec<u32>)> {
    let v: Vec<f32> = x.to_vec1()?;
    let mut idx: Vec<usize> = (0..v.len()).collect();
    idx.sort_by(|&a, &b| v[b].partial_cmp(&v[a]).unwrap_or(std::cmp::Ordering::Equal));
    idx.truncate(k);
    let vals: Vec<f32> = idx.iter().map(|&i| v[i]).collect();
    let ids: Vec<u32> = idx.iter().map(|&i| i as u32).collect();
    Ok((vals, ids))
}

fn sample_categorical(probs: &[f32], rng: &mut rand::rngs::ThreadRng) -> u32 {
    use rand::Rng;
    let total: f32 = probs.iter().filter(|x| x.is_finite()).sum();
    let mut r: f32 = rng.gen::<f32>() * total;
    for (i, &p) in probs.iter().enumerate() {
        if !p.is_finite() {
            continue;
        }
        r -= p;
        if r <= 0.0 {
            return i as u32;
        }
    }
    probs.len() as u32 - 1
}
```

```rust
//! Sliding-window next-token dataset (Ch. 3, "packing").

use anyhow::{anyhow, Result};
use candle_core::{Device, Tensor};

/// Yields (input, target) pairs of shape (seq_len,) / (seq_len,).
pub struct TextDataset {
    tokens: Vec<u32>,
    seq_len: usize,
    stride: usize,
    starts: Vec<usize>,
}

impl TextDataset {
    pub fn new(tokens: Vec<u32>, seq_len: usize, stride: Option<usize>) -> Result<Self> {
        let stride = stride.unwrap_or(seq_len);
        let n = tokens.len();
        if n < seq_len + 1 {
            return Err(anyhow!(
                "corpus too short ({} tokens) for seq_len={}",
                n,
                seq_len
            ));
        }
        let max_start = n - seq_len - 1;
        let starts: Vec<usize> = (0..=max_start).step_by(stride).collect();
        Ok(Self {
            tokens,
            seq_len,
            stride,
            starts,
        })
    }

    pub fn len(&self) -> usize {
        self.starts.len()
    }

    pub fn is_empty(&self) -> bool {
        self.starts.is_empty()
    }

    pub fn get(&self, idx: usize, dev: &Device) -> Result<(Tensor, Tensor)> {
        let s = self.starts[idx];
        let x: Vec<u32> = self.tokens[s..s + self.seq_len].to_vec();
        let y: Vec<u32> = self.tokens[s + 1..s + self.seq_len + 1].to_vec();
        let x = Tensor::from_vec(x, (self.seq_len,), dev)?;
        let y = Tensor::from_vec(y, (self.seq_len,), dev)?;
        Ok((x, y))
    }

    pub fn stride(&self) -> usize {
        self.stride
    }
}
```

```rust
//! Minimal next-token-prediction training loop (Chapters 11 & 12).

use anyhow::Result;
use candle_core::{DType, Device, Tensor};
use candle_nn::{loss, AdamW, Optimizer, ParamsAdamW, VarMap};

use crate::dataset::TextDataset;
use crate::model::Llm;

/// Training configuration.
pub struct TrainerConfig {
    pub lr: f64,
    pub weight_decay: f64,
    pub warmup_steps: usize,
    pub max_steps: usize,
    pub batch_size: usize,
    pub grad_clip: f64,
    pub log_every: usize,
}

impl Default for TrainerConfig {
    fn default() -> Self {
        Self {
            lr: 3e-4,
            weight_decay: 0.1,
            warmup_steps: 20,
            max_steps: 200,
            batch_size: 8,
            grad_clip: 1.0,
            log_every: 20,
        }
    }
}

/// Wraps the model + optimizer + LR schedule.
pub struct Trainer<'a> {
    pub model: &'a Llm,
    pub varmap: &'a VarMap,
    pub optimizer: AdamW,
    pub cfg: TrainerConfig,
    pub dev: Device,
}

impl<'a> Trainer<'a> {
    pub fn new(
        model: &'a Llm,
        varmap: &'a VarMap,
        cfg: TrainerConfig,
        dev: Device,
    ) -> Result<Self> {
        let params = ParamsAdamW {
            lr: cfg.lr,
            weight_decay: cfg.weight_decay,
            ..Default::default()
        };
        let optimizer = AdamW::new(varmap.all_vars(), params)?;
        Ok(Self {
            model,
            varmap,
            optimizer,
            cfg,
            dev,
        })
    }

    /// Train for `cfg.max_steps` iterations over the dataset.
    pub fn train(&mut self, train_ds: &TextDataset) -> Result<()> {
        let n = train_ds.len();
        let bs = self.cfg.batch_size;
        let mut step = 0usize;
        let mut idx = 0usize;
        let mut running_loss = 0.0f64;
        let mut running_n = 0usize;

        while step < self.cfg.max_steps {
            // Build a batch.
            let mut xs = Vec::with_capacity(bs);
            let mut ys = Vec::with_capacity(bs);
            for _ in 0..bs {
                let (x, y) = train_ds.get(idx % n, &self.dev)?;
                xs.push(x);
                ys.push(y);
                idx += 1;
            }
            let x = Tensor::stack(&xs, 0)?; // (B, S)
            let y = Tensor::stack(&ys, 0)?; // (B, S)

            // Forward.
            let (logits, _) = self.model.forward(&x, None, false)?;
            let (b, s, v) = logits.dims3()?;
            let logits_flat = logits.reshape((b * s, v))?;
            let y_flat = y.reshape((b * s,))?;

            let loss = loss::cross_entropy(&logits_flat, &y_flat)?;
            let loss_val: f32 = loss.to_scalar()?;

            // Backward + step.
            self.optimizer.backward_step(&loss)?;

            running_loss += loss_val as f64;
            running_n += 1;
            step += 1;

            if step % self.cfg.log_every == 0 {
                println!(
                    "[step {:>5}/{}] loss={:.4}",
                    step,
                    self.cfg.max_steps,
                    running_loss / running_n as f64
                );
                running_loss = 0.0;
                running_n = 0;
            }
        }

        let _ = DType::F32; // silence unused import when feature-gated
        Ok(())
    }
}
```

```rust
        ("greedy",   SamplingParams { do_sample: false, ..Default::default() }),
        ("temp=0.7", SamplingParams { do_sample: true, temperature: 0.7, top_k: 40, ..Default::default() }),
        ("top-p0.9", SamplingParams { do_sample: true, temperature: 0.9, top_p: 0.9, ..Default::default() }),
    ];

    for prompt in prompts {
        println!("\nPROMPT: {:?}", prompt);
        for (name, p) in &strategies {
            let out = gen.generate(prompt, p)?;
            println!("    [{:<10}] {:?}", name, out);
        }
    }

    // ---- 6. KV cache correctness --------------------------------------
    println!("\n[5/5] Verifying KV-cache correctness ...");
    let test_prompt = "the quick brown fox jumps";
    let ids = tokenizer.encode(test_prompt, true);
    let input = Tensor::from_vec(ids, (1, tokenizer.encode(test_prompt, true).len()), &dev)?;

    let (logits_no, _) = model.forward(&input, None, true)?;
    let last_no = logits_no.narrow(1, logits_no.dim(1)? - 1, 1)?.squeeze(1)?;

    let (_, caches) = model.forward(&input, None, true)?;
    let last_tok = input.narrow(1, input.dim(1)? - 1, 1)?;
    let (logits_c, _) = model.forward(&last_tok, caches.as_deref(), true)?;
    let last_c = logits_c.squeeze(1)?;

    let diff = (&last_no - &last_c)?.abs()?.max_all()?.to_scalar::<f32>()?;
    println!("    max |logits_no_cache - logits_with_cache| = {:.2e}", diff);
    if diff < 1e-3 {
        println!("    ✓ KV cache matches non-cached forward pass.");
    } else {
        println!("    ⚠ KV cache mismatch — check RoPE offsets and masking.");
    }

    println!("\n{}", "=".repeat(78));
    println!("Demo complete.");
    println!("{}", "=".repeat(78));
    Ok(())
}
```

## Enhancements Relative to the Python Version

| Feature | Python | Rust |
|---|---|---|
| **Static type checks** | runtime only | compile-time (traits, `Result`) |
| **Memory safety** | GC | ownership + borrow-checker, no GC |
| **Performance** | PyTorch dispatch overhead | zero-cost abstractions |
| **GQA** | runtime `repeat_interleave` | compile-time checked `broadcast_as` + `reshape` |
| **KV cache** | Python `Tuple[Tensor, Tensor]` | borrowed `Option<&KvCache>` — zero-copy reuse |
| **Errors** | exceptions | `anyhow::Result` propagation everywhere |
| **Serialization** | pickle | JSON (config, tokenizer) + `VarMap` safetensors |
| **Concurrency** | GIL-bound | full `Send + Sync`, multi-thread ready |
